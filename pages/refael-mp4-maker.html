ffmpeg one-liner (alternative, requires brew install ffmpeg) 
# ── single track (background.jpg + refael.mp3 in cwd) ──
ffmpeg -loop 1 -framerate 30 -i background.jpg -i refael.mp3 \
  -c:v libx264 -preset slow -crf 18 -pix_fmt yuv420p \
  -c:a aac -b:a 192k \
  -shortest -movflags +faststart \
  refael.mp4

# ── batch from a folder (self-contained, no pre-rendered covers) ──
# drop your .mp3 files into ./tracks/ first, then:
mkdir -p out
setopt -o NULL_GLOB 2>/dev/null || shopt -s nullglob  # zsh or bash

for f in tracks/*.mp3; do
  base="${f##*/}"; base="${base%.mp3}"
  ffmpeg -y -hide_banner -loglevel error \
    -f lavfi -i "color=c=0x0a0a14:s=1920x1080:r=30:d=86400" \
    -i "$f" \
    -vf "drawtext=text='${base}':fontcolor=0xe8e0d4:fontsize=140:x=(w-text_w)/2:y=(h-text_h)/2:font='Charter'" \
    -c:v libx264 -preset slow -crf 18 -pix_fmt yuv420p \
    -c:a aac -b:a 192k -shortest -movflags +faststart \
    "out/${base}.mp4"
done
# outputs land in ./out/<trackname>.mp4, one per input mp3
